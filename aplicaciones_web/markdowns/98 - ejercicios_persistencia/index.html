    32. Crear índice de palabras de un archivo

    Crea un script (indice_palabras.php) que lea un texto y cree un índice alfabético con cada palabra y su número de línea.

    <?php
$indice = [];
$puntero = fopen('texto.txt', 'r');
$numero = 1;

while (($linea = fgets($puntero)) !== false) {
    $palabras = preg_split('/\W+/u', strtolower($linea), -1, PREG_SPLIT_NO_EMPTY);

    foreach ($palabras as $palabra) {
        if (!isset($indice[$palabra])) {
            $indice[$palabra] = [];
        }
        if (!in_array($numero, $indice[$palabra])) {
            $indice[$palabra][] = $numero;
        }
    }

    $numero++;
}

fclose($puntero);
ksort($indice);

foreach ($indice as $palabra => $lineas) {
    echo $palabra . ": líneas " . implode(', ', $lineas) . "<br>";
}
?>

    <?php

header('Content-Type: application/json; charset=utf-8');

$metodo = $_SERVER['REQUEST_METHOD'];

try {
    $pdo = new PDO('mysql:host=localhost;dbname=empresa;charset=utf8mb4', 'root', 'tu_contrasena', [
        PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
        PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC
    ]);

    $coleccion = (new MongoDB\Client('mongodb://localhost:27017'))->tienda->productos;

    if ($metodo === 'GET') {
        $productos = $pdo->query("SELECT id, nombre, precio FROM productos")->fetchAll();

        $eventos = [];
        foreach ($coleccion->find([], ['limit' => 5]) as $documento) {
            $eventos[] = [
                'nombre' => $documento['nombre'],
                'precio' => $documento['precio']
            ];
        }

        echo json_encode(['mysql' => $productos, 'mongodb' => $eventos]);
    } elseif ($metodo === 'POST') {
        $datos = json_decode(file_get_contents('php://input'), true);

        $stmt = $pdo->prepare("INSERT INTO productos (nombre, precio) VALUES (:nombre, :precio)");
        $stmt->execute([':nombre' => $datos['nombre'], ':precio' => $datos['precio']]);

        $coleccion->insertOne(['nombre' => $datos['nombre'], 'precio' => $datos['precio']]);

        http_response_code(201);
        echo json_encode(['mensaje' => 'Producto creado']);
    } elseif ($metodo === 'PUT') {
        $datos = json_decode(file_get_contents('php://input'), true);

        $stmt = $pdo->prepare("UPDATE productos SET precio = :precio WHERE nombre = :nombre");
        $stmt->execute([':precio' => $datos['precio'], ':nombre' => $datos['nombre']]);

        $coleccion->updateOne(
            ['nombre' => $datos['nombre']],
            ['$set' => ['precio' => $datos['precio']]]
        );

        echo json_encode(['mensaje' => 'Producto actualizado']);
    } else {
        http_response_code(405);
        echo json_encode(['error' => 'Método no permitido']);
    }
} catch (Exception $e) {
    http_response_code(500);
    echo json_encode(['error' => $e->getMessage()]);
