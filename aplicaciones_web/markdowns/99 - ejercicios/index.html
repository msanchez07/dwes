    15. Creación de Clase SimpleValidator

    Crea una clase simple de validación (SimpleValidator) con los métodos __construct(), validate() y getErrors(). Implementa solo la regla required dentro del método validate() y haz que devuelva un array asociativo de errores.

    <?php
// SimpleValidator.php
class SimpleValidator {
    protected $data;
    protected $rules;
    protected $errors = [];

    public function __construct(array $data, array $rules) {
        $this->data = $data;
        $this->rules = $rules;
    }

    public function validate(): bool {
        foreach ($this->rules as $field => $rulesString) {
            $rules = explode('|', $rulesString);
            $value = $this->data[$field] ?? null;

            foreach ($rules as $rule) {
                if ($rule === 'required' && empty($value)) {
                    $this->addError($field, "El campo {$field} es obligatorio.");
                    continue;
                }
            }
        }

        return empty($this->errors);
    }

    protected function addError(string $field, string $message): void {
        if (!isset($this->errors[$field])) {
            $this->errors[$field] = $message;
        }
    }

    public function getErrors(): array {
        return $this->errors;
    }
}
?>

    16. Validación de Datos con Regex

    Añade la regla zip_code a tu clase SimpleValidator. Implementa la lógica para usar preg_match() y verificar que un campo llamado codigo_postal contenga exactamente 5 dígitos numéricos.

    <?php
// SimpleValidator.php
class SimpleValidator {
    protected $data;
    protected $rules;
    protected $errors = [];

    public function __construct(array $data, array $rules) {
        $this->data = $data;
        $this->rules = $rules;
    }

    public function validate(): bool {
        foreach ($this->rules as $field => $rulesString) {
            $rules = explode('|', $rulesString);
            $value = $this->data[$field] ?? null;

            foreach ($rules as $rule) {
                if ($rule === 'required' && empty($value)) {
                    $this->addError($field, "El campo {$field} es obligatorio.");
                    continue;
                }

                // Exactamente 5 dígitos numéricos.
                if ($rule === 'zip_code' && !preg_match('/^\d{5}$/', (string) $value)) {
                    $this->addError($field, "El campo {$field} debe tener 5 dígitos.");
                    continue;
                }
            }
        }

        return empty($this->errors);
    }

    protected function addError(string $field, string $message): void {
        if (!isset($this->errors[$field])) {
            $this->errors[$field] = $message;
        }
    }

    public function getErrors(): array {
        return $this->errors;
    }
}
?>

<?php
class SessionManager {
    // Inicia la sesión solo si no está iniciada todavía.
    public static function start(): void {
        if (session_status() === PHP_SESSION_NONE) {
            session_start();
        }
    }

    public static function set(string $key, $value): void {
        self::start();
        $_SESSION[$key] = $value;
    }

    public static function get(string $key, $default = null) {
        self::start();
        return $_SESSION[$key] ?? $default;
    }

    public static function has(string $key): bool {
        self::start();
        return isset($_SESSION[$key]);
    }

    public static function flash(string $key, $value): void {
        self::start();
        $_SESSION['_flash'][$key] = $value;
    }

    public static function getFlash(string $key, $default = null) {
        self::start();
        if (!isset($_SESSION['_flash'][$key])) {
            return $default;
        }
        $value = $_SESSION['_flash'][$key];
        unset($_SESSION['_flash'][$key]);
        return $value;
    }
}

    19. Inyección Segura de Datos para Contraseña

    Investiga la función password_needs_rehash(). Modifica tu procesador de login para que, si el login es correcto y la contraseña almacenada requiere rehash (debido a un algoritmo más antiguo o a cambios de configuración), automáticamente calcule un nuevo hash y simule su actualización en la "base de datos" ($usuarios).

    <?php
// procesar_login.php
session_start();

$usuarios = [
    'admin' => [
        'id' => 1,
        'password_hash' => '$2y$10$GLJg.PF/b0..y.Yk92d3A.29Jc5Qd3Y.E2rJ/NnKb.p2j/i5g7Ua.',
        'rol' => 'admin',
    ],
];

$usuario_form = $_POST['usuario'] ?? '';
$password_form = $_POST['password'] ?? '';

if (isset($usuarios[$usuario_form]) && password_verify($password_form, $usuarios[$usuario_form]['password_hash'])) {
    $hash_actual = $usuarios[$usuario_form]['password_hash'];

    // Si el hash guardado usa un algoritmo o coste obsoleto, lo
    // regeneramos aprovechando que conocemos la contraseña en claro.
    if (password_needs_rehash($hash_actual, PASSWORD_DEFAULT)) {
        $nuevo_hash = password_hash($password_form, PASSWORD_DEFAULT);
        // Simulamos la actualización en la "base de datos".
        $usuarios[$usuario_form]['password_hash'] = $nuevo_hash;
    }

    $_SESSION['usuario_id'] = $usuarios[$usuario_form]['id'];
    $_SESSION['rol'] = $usuarios[$usuario_form]['rol'];
    header('Location: privado.php');
    exit();
}

$_SESSION['error_login'] = "Usuario o contraseña incorrectos.";
header('Location: login.php');
exit();
?>
